# M3-10 Client Authn Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement M3-10 client authentication with a configurable static username/password authentication chain, while wiring an authorization chain into SUBSCRIBE/PUBLISH with current permit-all behavior.

**Architecture:** CONNECT keeps using `AuthnProvider` as the protocol-facing seam, but the concrete CDI implementation becomes `ConfiguredAuthnProvider`, backed by `AuthnChain` and resource-style authenticators. Authz mirrors the same shape: `DefaultProtocolEngine` depends on `AuthzProvider`, backed by `AuthzChain` and resource-style authorizers. Config files seed the current chains; future management UI can replace the resource source without changing protocol code.

**Implementation refinement:** The executed implementation upgraded the original boolean/enum authentication and authorization seams to rich results. `AuthnProvider` / `AuthnAuthenticator` return `AuthnResult`; `AuthzProvider` / `AuthzAuthorizer` return `AuthzResult`. Older snippets in this task plan preserve the original TDD steps, but the source of truth is the implemented result models.

**Tech Stack:** Java 21, Quarkus 3.33.1, SmallRye Config Mapping, Vert.x MQTT, JUnit 5, Maven.

```java
package io.github.vxmqmqtt.vxmq.authn;

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;

public class StaticPasswordAuthn implements AuthnAuthenticator {

    private final Map<String, String> passwordsByUsername;

    public StaticPasswordAuthn(Map<String, String> passwordsByUsername) {
        this.passwordsByUsername = Map.copyOf(new LinkedHashMap<>(
                Objects.requireNonNull(passwordsByUsername, "passwordsByUsername")));
    }

    @Override
    public AuthnResult authenticate(AuthnContext context) {
        String username = context.request().username();
        if (!passwordsByUsername.containsKey(username)) {
            return AuthnResult.ABSTAIN;
        }
        if (!context.request().passwordPresent()) {
            return AuthnResult.DENY;
        }
        return passwordsByUsername.get(username).equals(context.request().password())
                ? AuthnResult.ALLOW
                : AuthnResult.DENY;
    }
}
```

```java
package io.github.vxmqmqtt.vxmq.authn;

import java.util.List;

public class AuthnChain {

    private final List<AuthnDefinition> authenticators;
    private final AuthnNoMatchPolicy noMatchPolicy;

    public AuthnChain(
            List<AuthnDefinition> authenticators,
            AuthnNoMatchPolicy noMatchPolicy) {
        this.authenticators = List.copyOf(authenticators);
        this.noMatchPolicy = noMatchPolicy == null ? AuthnNoMatchPolicy.DENY : noMatchPolicy;
    }

    public static AuthnChain permitAll() {
        return new AuthnChain(
                List.of(new AuthnDefinition("permit-all", true, new PermitAllAuthnAuthenticator())),
                AuthnNoMatchPolicy.ALLOW);
    }

    public boolean allowConnect(AuthnContext context) {
        for (AuthnDefinition definition : authenticators) {
            if (!definition.enabled()) {
                continue;
            }
            AuthnResult decision = definition.authenticator().authenticate(context);
            if (decision == AuthnResult.ALLOW) {
                return true;
            }
            if (decision == AuthnResult.DENY) {
                return false;
            }
        }
        return noMatchPolicy.allow();
    }
}
```

Create `ConfiguredAuthnProvider`:

```java
package io.github.vxmqmqtt.vxmq.authn;

import io.github.vxmqmqtt.vxmq.protocol.model.ConnectRequest;
import io.github.vxmqmqtt.vxmq.transport.ClientConnection;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;

import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

@ApplicationScoped
public class ConfiguredAuthnProvider implements AuthnProvider {

    private final AuthnChain chain;

    @Inject
    public ConfiguredAuthnProvider(AuthnRuntimeConfig config) {
        this(buildChain(config));
    }

    public ConfiguredAuthnProvider(AuthnChain chain) {
        this.chain = chain;
    }

    @Override
    public boolean allowConnect(ClientConnection connection, ConnectRequest request) {
        return chain.allowConnect(new AuthnContext(connection, request));
    }

    private static AuthnChain buildChain(AuthnRuntimeConfig config) {
        if (config.authenticators().isEmpty()) {
            return AuthnChain.permitAll();
        }
        List<AuthnDefinition> definitions = config.authenticators().stream()
                .map(ConfiguredAuthnProvider::toDefinition)
                .toList();
        AuthnNoMatchPolicy noMatch = config.noMatch().orElse(AuthnNoMatchPolicy.DENY);
        return new AuthnChain(definitions, noMatch);
    }

    private static AuthnDefinition toDefinition(AuthnRuntimeConfig.AuthnAuthenticatorConfig config) {
        if (!"password".equals(config.mechanism()) || !"static".equals(config.backend())) {
            throw new IllegalArgumentException(
                    "Unsupported authenticator mechanism/backend: " + config.mechanism() + "/" + config.backend());
        }
        return new AuthnDefinition(
                config.id(),
                config.enabled(),
                new StaticPasswordAuthnAuthenticator(staticUsers(config)));
    }
```

```java
package io.github.vxmqmqtt.vxmq.authz;

import jakarta.enterprise.context.ApplicationScoped;

import java.util.List;

@ApplicationScoped
public class AuthzChain {

    private final List<AuthzDefinition> authorizers;
    private final AuthzNoMatchPolicy noMatchPolicy;

    public AuthzChain() {
        this(List.of(new AuthzDefinition("permit-all", true, new PermitAllAuthzAuthorizer())),
                AuthzNoMatchPolicy.ALLOW);
    }

    public AuthzChain(
            List<AuthzDefinition> authorizers,
            AuthzNoMatchPolicy noMatchPolicy) {
        this.authorizers = List.copyOf(authorizers);
        this.noMatchPolicy = noMatchPolicy == null ? AuthzNoMatchPolicy.ALLOW : noMatchPolicy;
    }

    public static AuthzChain permitAll() {
        return new AuthzChain(
                List.of(new AuthzDefinition("permit-all", true, new PermitAllAuthzAuthorizer())),
                AuthzNoMatchPolicy.ALLOW);
    }

    public boolean authorize(AuthzContext context) {
        for (AuthzDefinition definition : authorizers) {
            if (!definition.enabled()) {
                continue;
            }
            AuthzResult decision = definition.authorizer().authorize(context);
            if (decision == AuthzResult.ALLOW) {
                return true;
            }
            if (decision == AuthzResult.DENY) {
                return false;
            }
        }
        return noMatchPolicy.allow();
    }
}
```
